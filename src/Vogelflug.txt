public class VogelFlug {

    public static void main(String[] args) throws InterruptedException {
        String vogel = "\\ O /";
        int maxPosition = 20; // wie weit der Vogel fliegt

        while (true) {
            // von links nach rechts
            for (int i = 0; i <= maxPosition; i++) {
                printVogel(vogel, i);
                Thread.sleep(100);
            }

            // von rechts nach links
            for (int i = maxPosition; i >= 0; i--) {
                printVogel(vogel, i);
                Thread.sleep(100);
            }
        }
    }

    private static void printVogel(String vogel, int position) {
        System.out.print("\r"); // zurück an Zeilenanfang
        for (int i = 0; i < position; i++) {
            System.out.print(" ");
        }
        System.out.print(vogel);
    }
}
